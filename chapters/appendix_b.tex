\chapter{Complete 30-Feature Behavioral Extraction Specification}
\label{app:features}

Table~\ref{tab:full_features} presents the complete specification of the 30 behavioral features used for OCEAN trait estimation. Features~1--22 are the original psycholinguistic markers; Features~23--30 were added to address Conscientiousness (C), Openness (O), Neuroticism (N), and Agreeableness (A) expression fidelity.

\begin{table}[ht]
\centering
\caption{Complete 30-Feature Behavioral Extraction Specification}
\label{tab:full_features}
\small
\begin{tabular}{rlp{5.8cm}l}
\toprule
\# & Feature & Extraction Method & Primary Trait \\
\midrule
\multicolumn{4}{l}{\textit{Volume \& Verbosity}} \\
1  & \texttt{avg\_words\_per\_turn}    & Mean word count across turns & E \\
2  & \texttt{max\_words\_in\_turn}     & Peak word count in any turn & E \\
3  & \texttt{min\_words\_in\_turn}     & Minimum word count in any turn & E \\
4  & \texttt{word\_count\_variance}    & Std.\ dev.\ of word counts & O, N \\
\midrule
\multicolumn{4}{l}{\textit{Social Engagement}} \\
5  & \texttt{question\_ratio}          & Turns ending with ``?'' / total & E, O \\
6  & \texttt{exclamation\_ratio}       & Turns with ``!'' / total & E \\
7  & \texttt{name\_mention\_count}     & Occurrences of agent names & E, A \\
8  & \texttt{turn\_initiation\_ratio}  & Turns initiating after non-self / total & E \\
9  & \texttt{avg\_response\_latency}   & Avg.\ position after prior non-self turn & E (inv.) \\
\midrule
\multicolumn{4}{l}{\textit{Language Style}} \\
10 & \texttt{hedge\_count}             & ``maybe'', ``perhaps'', ``might be'' \etal & E, N \\
11 & \texttt{certainty\_count}         & ``definitely'', ``absolutely'', ``clearly'' \etal & E, N (inv.) \\
12 & \texttt{first\_person\_ratio}     & \{I, me, my, mine, myself\} / total words & N \\
13 & \texttt{inclusive\_pronoun\_ratio} & \{we, our, us\} / total words & A \\
14 & \texttt{unique\_word\_ratio}       & Unique words / total words & O \\
15 & \texttt{long\_sentence\_ratio}    & Sentences $>$20 words / total sentences & O, C \\
\midrule
\multicolumn{4}{l}{\textit{Behavioral Signals}} \\
16 & \texttt{disagreement\_count}      & ``disagree'', ``however'', ``push back'' \etal (17 phrases) & Low A \\
17 & \texttt{acknowledgment\_count}    & ``good point'', ``agree'', ``exactly'' \etal (15 phrases) & A / Syc. \\
18 & \texttt{idea\_count}              & ``what if'', ``we could'', ``how about'' \etal (14 phrases) & O \\
19 & \texttt{planning\_count}          & ``first'', ``step'', ``plan'', ``prioritize'' \etal (10 phrases) & C \\
20 & \texttt{conditional\_ratio}       & Regex: ``if\ldots then'', ``should'', ``assuming'' \etal & C \\
21 & \texttt{emotional\_word\_count}   & ``worried'', ``anxious'', ``frustrated'' \etal (10 words) & N, Low C \\
22 & \texttt{positive\_emotion\_count} & ``excited'', ``wonderful'', ``fantastic'' \etal (10 words) & A, Low N \\
\midrule
\multicolumn{4}{l}{\textit{Added Features (Critique Response)}} \\
23 & \texttt{structure\_marker\_count} & Numbered lists, ``firstly/secondly'', ``in summary'' \etal & C \\
24 & \texttt{reference\_back\_count}   & ``as we discussed'', ``going back to'' \etal (14 phrases) & C \\
25 & \texttt{action\_item\_count}      & ``who will'', ``deadline'', ``next step'' \etal (12 phrases) & C \\
26 & \texttt{hypothetical\_count}      & ``hypothetically'', ``thought experiment'' \etal (10 phrases) & O \\
27 & \texttt{apology\_count}           & ``I apologize'', ``my fault'', ``my mistake'' \etal (9 phrases) & N \\
28 & \texttt{self\_doubt\_count}       & ``I'm probably wrong'', ``I could be mistaken'' \etal (12 phrases) & N \\
29 & \texttt{reassurance\_seeking}     & ``does that make sense'', ``am I on track'' \etal (10 phrases) & N \\
30 & \texttt{negation\_count}          & \{no, not, won't, can't, shouldn't, \ldots\} (18 words) & Low A \\
\bottomrule
\end{tabular}
\end{table}
